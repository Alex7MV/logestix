find_package(Threads REQUIRED)

# ── Phase 5: embed the TaskMosaic CE SPA (www/) as immutable asset bytes ──────
# Each file in www/ is read as HEX and emitted into a generated header as a
# 64-byte cache-line aligned `std::uint8_t[]`, plus an inline asset table and
# lookup. The compiled bundle is served straight from memory — zero disk I/O and
# zero per-request allocation on the static serving path.
set(_lsxwww_root "${CMAKE_CURRENT_SOURCE_DIR}/www")

set(_lsxwww_files
    "${_lsxwww_root}/index.html"
    "${_lsxwww_root}/css/styles.css"
    "${_lsxwww_root}/js/app.js"
    "${_lsxwww_root}/js/modules/mosaic.js"
    "${_lsxwww_root}/js/modules/renderer.js"
    "${_lsxwww_root}/js/modules/stream.js"
)
set(_lsxwww_paths
    "/"  "/css/styles.css"  "/js/app.js"
    "/js/modules/mosaic.js" "/js/modules/renderer.js" "/js/modules/stream.js"
)
set(_lsxwww_names kRoot kStyles kApp kMosaic kRenderer kStream)
set(_lsxwww_mimes
    "text/html; charset=utf-8"
    "text/css; charset=utf-8"
    "text/javascript; charset=utf-8"
    "text/javascript; charset=utf-8"
    "text/javascript; charset=utf-8"
    "text/javascript; charset=utf-8"
)

set(_lsxwww_out "${CMAKE_CURRENT_BINARY_DIR}/generated/lsx_embedded_www.h")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/generated")

set(_hdr "#pragma once\n// AUTO-GENERATED by libs/lsxnetwork/CMakeLists.txt — DO NOT EDIT.\n#include <cstddef>\n#include <cstdint>\n#include <string_view>\n#include \"lsxnetwork/spa_assets.h\"\nnamespace lsxnetwork { namespace spa { namespace gen {\n")

set(_idx 0)
foreach(_f ${_lsxwww_files})
    list(GET _lsxwww_names ${_idx} _nn)
    file(READ "${_f}" _hex HEX)
    string(REGEX MATCHALL ".." _bytes "${_hex}")
    string(APPEND _hdr "alignas(64) const std::uint8_t ${_nn}[] = {")
    foreach(_b ${_bytes})
        string(APPEND _hdr "0x${_b},")
    endforeach()
    string(APPEND _hdr "};\n")
    math(EXPR _idx "${_idx}+1")
endforeach()
string(APPEND _hdr "}  // namespace gen\ninline const Asset* GetSpaAssets() {\nstatic const Asset kAssets[] = {\n")
set(_idx 0)
foreach(_nn ${_lsxwww_names})
    list(GET _lsxwww_paths ${_idx} _pp)
    list(GET _lsxwww_mimes ${_idx} _mm)
    string(APPEND _hdr "    {\"${_pp}\", gen::${_nn}, sizeof(gen::${_nn}), \"${_mm}\"},\n")
    math(EXPR _idx "${_idx}+1")
endforeach()
string(APPEND _hdr "};\nreturn kAssets;\n}\ninline std::size_t GetSpaAssetCount() { return ${_idx}; }\ninline const Asset* FindSpaAsset(std::string_view path) {\nconst Asset* a = GetSpaAssets();\nconst std::size_t n = GetSpaAssetCount();\nfor (std::size_t i = 0; i < n; ++i) { if (path == a[i].path) return &a[i]; }\nreturn nullptr;\n}\n}\n}  // namespace lsxnetwork::spa\n")

file(WRITE "${_lsxwww_out}" "${_hdr}")
unset(_hdr)

add_library(lsxnetwork STATIC
    src/affinity.cpp
    src/anthropic_endpoint.cpp
    src/fabric_stream_client.cpp
    src/fabric_wire.cpp
    src/frame_queue.cpp
    src/http_router.cpp
    src/http_server.cpp
    src/io_worker.cpp
    src/lsx_token_translator.cpp
    src/openai_endpoint.cpp
    src/zmq_context.cpp
)

target_include_directories(lsxnetwork
    PUBLIC
        "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>"
        "$<BUILD_INTERFACE:${CMAKE_CURRENT_BINARY_DIR}/generated>"
)

# simdjson is referenced by lsxnetwork's public header (openai_endpoint.h), so it
# must be PUBLIC. libzmq-static and Threads must be PUBLIC as well: lsxnetwork is
# a STATIC library, so its link dependencies are only re-exported to consumers
# through the interface (otherwise zmq_*/pthread symbols go unresolved in the
# final executable).
target_link_libraries(lsxnetwork PUBLIC
    simdjson::simdjson
    libzmq-static
    Threads::Threads
    absl::flat_hash_map
    absl::btree
)

# spdlog is header-only (FetchContent default), so it cannot leak link symbols;
# PRIVATE keeps its include dirs out of the public interface.
target_link_libraries(lsxnetwork PRIVATE
    spdlog::spdlog
    httplib::httplib
)

# Static libzmq on Windows depends on Winsock/IP helper libraries.
if(WIN32)
    target_link_libraries(lsxnetwork PUBLIC
        ws2_32
        iphlpapi
    )
endif()
