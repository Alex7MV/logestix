**Goal:** Raise Kimi-K2.7-Code default decode throughput on RTX 5090 from the measured baseline (Phase 0) to >= 100 tok/s via a batched tensor-core routed-expert chain, dual-stream expert prefetch and device-side token feed.

**Architecture:** The TC batch entry points live in the (renamed) model-neutral `lsx_tc_gemm` layer; Kimi consumes them through a new `RunKimiRoutedTcBatch` launcher with a host-built 8-entry slot table. Spec: `docs/superpowers/specs/2026-09-08-kimi-100tps-design.md` (incl. binding amendments: rename prerequisite, host-built slot table, BF16 act-quant, kimi-scoped VRAM reserve).

**Tech Stack:** CUDA 13.3 (sm_120a/sm_121a `mxf4nvf4` warp-level MMA), C++20/CMake presets, Catch2 v3, Linux AOCC toolchain (`build/linux-release-znver4-aocc`).

```cpp
__global__ void ActQuantBf16Kernel(const __nv_bfloat16 *__restrict__ x,
                                   int64_t n, uint8_t *__restrict__ packed,
                                   uint8_t *__restrict__ scales)
{
    const int64_t blk = static_cast<int64_t>(blockIdx.x) * blockDim.x +
                        threadIdx.x;
    const int64_t base = blk * 16;
    if (base >= n) return;
    const int len =
        static_cast<int>(min(static_cast<long long>(16),
                             static_cast<long long>(n - base)));
    float absmax = 0.0f;
    for (int i = 0; i < len; ++i) {
        const float v = fabsf(__bfloat162float(x[base + i]));
        absmax = fmaxf(absmax, v);
    }
    const uint8_t sb =
        lsxcommon::nvfp4::FloatToE4m3P2(absmax * (1.0f / 6.0f));
    scales[blk] = sb;
    const float inv_s = gpu::nvfp4::E4M3ScaleToFloat(sb);
    const float inv = inv_s > 0.0f ? 1.0f / inv_s : 0.0f;
    for (int i = 0; i < len; i += 2) {
        const uint8_t lo =
            kernels::QuantizeE2m1TieUp(__bfloat162float(x[base + i]) * inv);
        const uint8_t hi =
            (i + 1 < len)
                ? kernels::QuantizeE2m1TieUp(
                      __bfloat162float(x[base + i + 1]) * inv)
                : 0;
        packed[(base + i) / 2] = static_cast<uint8_t>((hi << 4) | lo);
    }
}

cudaError_t LsxTcQuantizeActsNvfp4Bf16(cudaStream_t stream,
                                       const __nv_bfloat16 *x, int seq,
                                       int K, uint8_t *d_packed,
                                       uint8_t *d_scales)
{
    if (!x || !d_packed || !d_scales || seq <= 0 || (K & 15) != 0)
        return cudaErrorInvalidValue;
    const int64_t n = static_cast<int64_t>(seq) * K;
    ActQuantBf16Kernel<<<static_cast<int>((n + 255) / 256), 256, 0, stream>>>(
        x, n, d_packed, d_scales);
    return cudaGetLastError();
}
```

```cpp
HostProj MakeProj(int seed, int64_t rows, int64_t cols)
{
    HostProj p;
    const int64_t numel = rows * cols;
    p.packed.assign(static_cast<size_t>(numel / 2), 0);
    p.scales.assign(static_cast<size_t>(numel / 16), 0);
    std::mt19937 rng(static_cast<unsigned>(seed));
    std::uniform_int_distribution<int> nib(0, 15);
    std::uniform_int_distribution<int> sc(-4, -1);
    for (int64_t i = 0; i < numel; ++i) {
        const int v = nib(rng);
        p.packed[static_cast<size_t>(i >> 1)] |=
            static_cast<uint8_t>(v << ((i & 1) ? 4 : 0));
        if ((i & 15) == 0)
            p.scales[static_cast<size_t>(i >> 4)] =
                static_cast<uint8_t>(((sc(rng) + 7) & 0xF) << 3);
    }
    return p;
}
```

```cpp
namespace {

struct KimiTcScratchLayout {
    uint8_t *xq;        uint8_t *xqs;
    __nv_bfloat16 *ygu;         // [K][2I]
    __nv_bfloat16 *hbf;         // [K][I]  fp16 view of swiglu output
    uint8_t *hq;        uint8_t *hqs;
    __half *yp;                 // [K][H]
    float *splitk13;            // sk*K*2I (may be null when sk==1)
    float *splitk2;             // sk*K*H (may be null when sk==1)
    float *acc;                 // [H] fp32 accumulator
};

KimiTcScratchLayout LayoutKimiTcScratch(void *base, size_t bytes, int sk)
{
    auto align = [](size_t v) { return (v + 255) / 256 * 256; };
    constexpr int I = kKimiMoeIntermediate;
    constexpr int H = kKimiHidden;
    constexpr int K = kKimiTopK;
    KimiTcScratchLayout L{};
    uint8_t *p = static_cast<uint8_t *>(base);
    auto take = [&](size_t n) { uint8_t *r = p; p += align(n); return r; };
    L.xq  = take(static_cast<size_t>(H) / 2);
    L.xqs = take(static_cast<size_t>(H) / 16);
    L.ygu = reinterpret_cast<__nv_bfloat16 *>(take(
        static_cast<size_t>(K) * 2 * I * sizeof(__nv_bfloat16)));
    L.hbf = reinterpret_cast<__nv_bfloat16 *>(take(
        static_cast<size_t>(K) * I * sizeof(__nv_bfloat16)));
    L.hq  = take(static_cast<size_t>(K) * I / 2);
    L.hqs = take(static_cast<size_t>(K) * I / 16);
    L.yp  = reinterpret_cast<__half *>(take(
        static_cast<size_t>(K) * H * sizeof(__half)));
    L.splitk13 = sk > 1 ? reinterpret_cast<float *>(take(
        static_cast<size_t>(sk) * K * 2 * I * sizeof(float))) : nullptr;
    L.splitk2  = sk > 1 ? reinterpret_cast<float *>(take(
        static_cast<size_t>(sk) * K * H * sizeof(float))) : nullptr;
    L.acc = reinterpret_cast<float *>(take(H * sizeof(float)));

    return L;

    constexpr int H = kKimiHidden;
    constexpr int K = kKimiTopK;

    b += align(static_cast<size_t>(H) / 16);

    b += align(static_cast<size_t>(K) * I * sizeof(__nv_bfloat16));
    b += align(static_cast<size_t>(K) * I / 2);

    if (sk2 > 1)

cudaError_t RunKimiRoutedTcBatch(cublasHandle_t, cudaStream_t stream,
                                 const uint8_t *pack_base,
                                 const uint8_t *scale_base,
                                 int64_t pack_stride, int64_t scale_stride,
                                 const lsxcommon::lsx_tc::LsxTcBatchSlot *d_tbl,
                                 int K, const __nv_bfloat16 *d_x,
                                 __nv_bfloat16 *d_out, void *d_scratch,
                                 size_t scratch_bytes)
{
    namespace tc = lsxcommon::lsx_tc;
    constexpr int I = kKimiMoeIntermediate;
    constexpr int H = kKimiHidden;
    if (!pack_base || !scale_base || !d_tbl || !d_x || !d_out || !d_scratch ||
        K <= 0 || K > kKimiTopK || (I & 15) != 0 ||
        scratch_bytes < KimiTcBatchScratchBytes())
        return cudaErrorInvalidValue;

    const int sk13 = tc::LsxTcBatchSplitK(H, 2 * I);
    const int sk2 = tc::LsxTcBatchSplitK(I, H);
    const KimiTcScratchLayout s =
        LayoutKimiTcScratch(d_scratch, scratch_bytes, std::max(sk13, sk2));

    cudaError_t e = tc::LsxTcQuantizeActsNvfp4Bf16(stream, d_x, 1, H,
                                                   s.xq, s.xqs);

    // unnecessary: the TC kernels accept caller-quantized activations
    // (ap/as) directly — no fp16 input is needed on this path.
    e = tc::RunLsxTcGemmBatch13(stream, s.xq, s.xqs, pack_base, scale_base,
                                pack_stride, scale_stride, d_tbl, K, H,

                                reinterpret_cast<__half *>(s.ygu),
                                s.splitk13);

    // swiglu-quant shape but source bf16 gate/up. Implemented here as a
    // bf16 twin of qwen's moe_swiglu_quant_batch_kernel (not exported).
    //gate/up come out of the w13 GEMM as fp16 in ygu.
    {
        const int64_t groups =
            static_cast<int64_t>(K) * ((I + 15) / 16);
        kimi_tc_swiglu_quant_kernel<<<static_cast<int>((groups + 255) / 256),
                                      256, 0, stream>>>(
            reinterpret_cast<const __half *>(s.ygu), K, I,
            reinterpret_cast<__half *>(s.hbf), s.hq, s.hqs);
        if (cudaGetLastError() != cudaSuccess)
            return cudaGetLastError();

            s.yp, d_tbl, K, H, s.acc, d_out);
        return cudaGetLastError();
    }
}
```

```cpp
// Slot-major swiglu + NVFP4 quant (bf16-source twin of the qwen
// moe_swiglu_quant_batch_kernel math; fp16 gate/up come from the w13 TC
// GEMM). Same per-16 codec: FloatToE4m3P2 + QuantizeE2m1TieUp.
__global__ void kimi_tc_swiglu_quant_kernel(
    const __half *__restrict__ ygu, int k_slots, int I,
    __half *__restrict__ h, uint8_t *__restrict__ packed,
    uint8_t *__restrict__ scales)
{
    const int64_t groups_per_slot = (I + 15) / 16;
    const int64_t g = static_cast<int64_t>(blockIdx.x) * blockDim.x +
                      threadIdx.x;
    if (g >= static_cast<int64_t>(k_slots) * groups_per_slot) return;
    const int s = static_cast<int>(g / groups_per_slot);
    const int j0 = static_cast<int>((g - static_cast<int64_t>(s) *
                                              groups_per_slot) * 16);
    const int len = (j0 + 16 <= I) ? 16 : (I - j0);
    float vals[16];
    float absmax = 0.0f;
    for (int i = 0; i < len; ++i) {
        const float gv = __half2float(
            ygu[static_cast<int64_t>(s) * 2 * I + j0 + i]);
        const float uv = __half2float(
            ygu[static_cast<int64_t>(s) * 2 * I + I + j0 + i]);
        const float sig = __fdiv_rn(1.0f, __fadd_rn(1.0f, expf(-gv)));
        const __half hv = __float2half(__fmul_rn(sig, __fmul_rn(gv, uv)));
        h[static_cast<int64_t>(s) * I + j0 + i] = hv;
        vals[i] = __half2float(hv);
        absmax = fmaxf(absmax, fabsf(vals[i]));
    }
    const uint8_t sb =
        lsxcommon::nvfp4::FloatToE4m3P2(absmax * (1.0f / 6.0f));
    scales[g] = sb;
    const float inv_s = gpu::nvfp4::E4M3ScaleToFloat(sb);
    const float inv = inv_s > 0.0f ? 1.0f / inv_s : 0.0f;
    const int64_t slot_base = static_cast<int64_t>(s) * I;
    for (int i = 0; i < len; i += 2) {
        const uint8_t lo =
            lsxcommon::kernels::QuantizeE2m1TieUp(vals[i] * inv);
        const uint8_t hi = (i + 1 < len)
                               ? lsxcommon::kernels::QuantizeE2m1TieUp(
                                     vals[i + 1] * inv)
                               : 0;
        packed[(slot_base + j0 + i) / 2] =
            static_cast<uint8_t>((hi << 4) | lo);
    }
}

// Ascending-k deterministic reduce: acc starts at zero, k=0..K-1 in order,
// then a single bf16 store (legacy chain accumulates bf16 per-expert; the
// TC path keeps the fp32 accumulator until the end — tolerance class).
__global__ void kimi_tc_reduce_kernel(const __half *__restrict__ yp,
                                      const lsxcommon::lsx_tc::LsxTcBatchSlot *__restrict__ tbl,
                                      int k_slots, int n,
                                      float *__restrict__ acc,
                                      __nv_bfloat16 *__restrict__ out)
{
    const int j = blockIdx.x * blockDim.x + threadIdx.x;
    if (j >= n) return;
    float t = 0.0f;
    for (int k = 0; k < k_slots; ++k) {
        const lsxcommon::lsx_tc::LsxTcBatchSlot b = tbl[k];
        t = fmaf(b.coeff, __half2float(
              yp[static_cast<int64_t>(k) * n + j]), t);
    }
    out[j] = __float2bfloat16_rn(t);
}
```

Rationale: after `KimiRouteFinish` for layer l, the host knows the top-8; the SAME route repeats with high probability at step+1. Enqueue (right after the TC batch launch of layer l, i.e. while the GPU still has queued compute) an LRU-noop + upload for the NEXT layer using the CURRENT h_ids:

```cpp

                    // persistence across decode steps). Resolve is a host

                                db, *cache, pb, l + 1, kKimiNumRouted);
```

CAUTION (correctness): `KimiMoeResolveBatch` advances the LRU clock and may ASSIGN slots + evict. The layer l+1 router will resolve its own batch again at step-processing time; the double-resolve is LRU-consistent (second call refreshes last_use; assignments are idempotent for residents).
