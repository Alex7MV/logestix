`tools/kimi_rope_capture.py`:

```python
#!/usr/bin/env python3
"""Capture Kimi-K2.7-Code RoPE + attention-score goldens from the vendored
reference implementation (modeling_deepseek.py shipped with the checkpoint).

The vendor's apply_rotary_pos_emb de-interleaves with
view(d//2, 2).transpose().reshape() before rotate_half, i.e. the effective
rotation is on ADJACENT pairs (2i, 2i+1) with inv_freq[i]. The function
returns the permuted layout, so the output is un-permuted here to the natural
element order before it is written to the fixture.

Usage:
  python3 tools/kimi_rope_capture.py --model /mnt/fast_storage/lsx_model/moonshotai/Kimi-K2.7-Code \
      --out tests/fixtures/kimi/rope_attn.txt
"""
import argparse
import importlib
import os
import shutil
import sys
import tempfile

import torch


def load_vendored(model_dir):
    root = tempfile.mkdtemp(prefix="kimi_vendor_")
    pkg = os.path.join(root, "kimi_vendor")
    os.makedirs(pkg)
    for name in ("modeling_deepseek.py", "configuration_deepseek.py"):
        shutil.copy(os.path.join(model_dir, name), pkg)
    open(os.path.join(pkg, "__init__.py"), "w").close()
    sys.path.insert(0, root)
    return importlib.import_module("kimi_vendor.modeling_deepseek")


def unpermute(y):
    # y = [x0,x2,...,x62, x1,x3,...,x63] -> natural [x0,x1,...,x63]
    n = y.shape[-1]
    out = torch.empty_like(y)
    out[..., 0::2] = y[..., : n // 2]
    out[..., 1::2] = y[..., n // 2 :]
    return out


def fmt(vals):
    return " ".join("%.9g" % float(v) for v in vals)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--model", required=True)
    ap.add_argument("--out", required=True)
    args = ap.parse_args()

    m = load_vendored(args.model)
    torch.manual_seed(20260824)
    rope = m.DeepseekV3YarnRotaryEmbedding(
        64, max_position_embeddings=262144, base=50000.0,
        scaling_factor=64.0, original_max_position_embeddings=4096,
        beta_fast=32.0, beta_slow=1.0, mscale=1.0, mscale_all_dim=1.0)
    seq = 9
    cos, sin = rope(torch.zeros(1, 1, seq, 64), seq_len=seq)
    q = torch.randn(1, 1, seq, 64)
    k = torch.randn(1, 1, seq, 64)
    pos = torch.arange(seq).view(1, seq)
    qr, kr = m.apply_rotary_pos_emb(q, k, cos, sin, position_ids=pos)
    qn = unpermute(qr)
    kn = unpermute(kr)

    lines = ["rope_dim 64", "positions %d" % seq,
             "inv_freq " + fmt(rope.inv_freq.tolist())]
    for p in range(seq):
        lines.append("q_raw %d %s" % (p, fmt(q[0, 0, p].tolist())))
        lines.append("k_raw %d %s" % (p, fmt(k[0, 0, p].tolist())))
        lines.append("q_roped %d %s" % (p, fmt(qn[0, 0, p].tolist())))
        lines.append("k_roped %d %s" % (p, fmt(kn[0, 0, p].tolist())))
        dot = float((qn[0, 0, p] * kn[0, 0, p]).sum())
        lines.append("dot %d %.9g" % (p, dot))
    os.makedirs(os.path.dirname(args.out), exist_ok=True)
    with open(args.out, "w", encoding="utf-8") as fh:
        fh.write("\n".join(lines) + "\n")
    print("wrote", args.out)


if __name__ == "__main__":
    main()
```

```cpp
// YaRN attention scale: the reference multiplies q_head_dim**-0.5 by
// yarn_get_mscale(factor, mscale_all_dim)**2 when mscale_all_dim is truthy.
// For this config (factor 64, mscale = mscale_all_dim = 1.0):
//   mscale = 0.1*ln(64) + 1 = 1.4158883083359672
// The cos/sin path is unscaled because yarn_get_mscale(64, 1)/itself == 1.
constexpr float kKimiYarnMscale = 1.4158883083359672f;

// One adjacent-pair RoPE rotation: element pair (2p, 2p+1) by theta.
// Single source of truth for the kernel and the parity test; the golden is
// captured from the checkpoint's own modeling_deepseek.py.
__device__ __host__ inline void KimiRopePair(int p, float theta, float *r)
{
    const int a = p;               // RED: half-split pairs (i, i+32)
    const int b = p + kKimiQkRope / 2;
    const float cs = cosf(theta), sn = sinf(theta);
    const float x = r[a], y = r[b];
    r[a] = x * cs - y * sn;
    r[b] = x * sn + y * cs;
}

// YaRN interpolation over the pair index (HF DeepseekV3YarnRotaryEmbedding).
std::vector<float> KimiYarnInvFreq();

// Attention temperature used by the MLA kernels. Header-inline because the
// decode kernel calls it from device code and the parity test calls it from
// the host. (Task 3 adds the YaRN mscale**2 factor; this commit keeps the
// current value so the parity test fails on the number, not on the build.)
__device__ __host__ inline float KimiAttnScale()
{
    return rsqrtf(static_cast<float>(kKimiQkNope + kKimiQkRope));
}
```

```cpp
std::vector<float> KimiYarnInvFreq()
{
    const double dim = kKimiQkRope;
    const double base = kKimiRopeTheta;
    const double max_pos = kKimiYarnOriginalMaxPos;
    const double low =
        std::floor(YarnCorrectionDim(32.0, dim, base, max_pos));
    const double high =
        std::ceil(YarnCorrectionDim(1.0, dim, base, max_pos));
    std::vector<float> inv_freq(kRopePairs);
    for (int i = 0; i < kRopePairs; ++i) {
        const double extra = std::pow(base, -2.0 * i / dim);
        const double inter = extra / kKimiYarnFactor;
        double ramp = (high != low)
                          ? (static_cast<double>(i) - low) / (high - low)
                          : 1.0;
        ramp = std::min(std::max(ramp, 0.0), 1.0);
        inv_freq[i] = static_cast<float>(inter * ramp + extra * (1.0 - ramp));
    }
    return inv_freq;
}
```

```cpp
// Pins the MLA RoPE and attention temperature to the checkpoint's own
// reference (modeling_deepseek.py). The fixture is captured by
// tools/kimi_rope_capture.py; it is NOT produced by these helpers, so a
// regression cannot be self-consistent.
#include <catch2/catch_test_macros.hpp>

#include <cmath>
#include <cstdio>
#include <cstdlib>
#include <map>
#include <string>
#include <vector>

#include "lsxcommon/kimi_moe_load.h"

namespace {

struct Fixture {
    int rope_dim = 0;
    int positions = 0;
    std::vector<float> inv_freq;
    std::map<int, std::vector<float>> q_raw, k_raw, q_roped, k_roped;
    std::map<int, float> dot;
};

float NextWord(FILE *f)
{
    char buf[64];
    REQUIRE(std::fscanf(f, "%63s", buf) == 1);
    return std::strtof(buf, nullptr);
}

Fixture LoadFixture()
{
    FILE *f = std::fopen(LSX_KIMI_FIXTURE_DIR "/rope_attn.txt", "rb");
    REQUIRE(f != nullptr);
    Fixture x;
    char key[64];
    while (std::fscanf(f, "%63s", key) == 1) {
        const std::string k(key);
        if (k == "rope_dim") {
            x.rope_dim = static_cast<int>(NextWord(f));
        } else if (k == "positions") {
            x.positions = static_cast<int>(NextWord(f));
        } else if (k == "inv_freq") {
            x.inv_freq.resize(x.rope_dim / 2);
            for (auto &v : x.inv_freq) v = NextWord(f);
        } else if (k == "q_raw" || k == "k_raw" || k == "q_roped" ||
                   k == "k_roped") {
            const int p = static_cast<int>(NextWord(f));
            std::vector<float> v(x.rope_dim);
            for (auto &e : v) e = NextWord(f);
            if (k == "q_raw") x.q_raw[p] = v;
            else if (k == "k_raw") x.k_raw[p] = v;
            else if (k == "q_roped") x.q_roped[p] = v;
            else x.k_roped[p] = v;
        } else if (k == "dot") {
            const int p = static_cast<int>(NextWord(f));
            x.dot[p] = NextWord(f);
        } else {
            FAIL("unexpected fixture key " << k);
        }
    }
    std::fclose(f);
    REQUIRE(x.rope_dim == 64);
    REQUIRE(x.positions == 9);
    REQUIRE(x.inv_freq.size() == 32);
    return x;

TEST_CASE("kimi MLA rope matches the vendored interleaved implementation",
```

```python
#!/usr/bin/env python3
"""Capture Kimi token ids from the checkpoint's own tokenizer (tiktoken).

The reference encode path is `tok(text)` (== `tok.encode(text)` ==
`tok(text, add_special_tokens=False)`): no BOS/EOS is added. The fixture is
binary-safe (length-prefixed), so newlines and markers survive.

Usage:
  python3 tools/kimi_tokenizer_capture.py \
      --model /mnt/fast_storage/lsx_model/moonshotai/Kimi-K2.7-Code \
      --corpus tests/fixtures/kimi/tokenizer_corpus.txt \
      --out tests/fixtures/kimi/tokenizer_parity.txt
"""
import argparse
import os

from transformers import AutoTokenizer


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--model", required=True)
    ap.add_argument("--corpus", required=True)
    ap.add_argument("--out", required=True)
    args = ap.parse_args()
    tok = AutoTokenizer.from_pretrained(args.model, trust_remote_code=True)

    cases = []
    with open(args.corpus, encoding="utf-8") as fh:
        for line in fh.read().split("\n"):
            if not line or line.startswith("#"):
                continue
            cases.append(line)

    os.makedirs(os.path.dirname(args.out), exist_ok=True)
    with open(args.out, "wb") as out:
        for text in cases:
            ids = tok(text).input_ids
            raw = text.encode("utf-8")
            out.write(("CASE %d %d %s\n" % (
                len(raw), len(ids), " ".join(str(i) for i in ids))).encode())
            out.write(raw + b"\n")
    print("wrote", args.out, "cases:", len(cases))


if __name__ == "__main__":
    main()
```

```cpp
#pragma once

// Kimi-K2.7-Code native conversation + tool-call codec (host-only).
// Grammar transcribed from the checkpoint's chat_template.jinja:
//   <|im_system|>{name}<|im_middle|>{text}<|im_end|>
//   <|im_user|>{name}<|im_middle|>{text}<|im_end|>
//   <|im_assistant|>{name}<|im_middle|><think>{reasoning}</think>{content}
//       {tool-calls}<|im_end|>
//   tool result: <|im_system|>tool<|im_middle|>## Return of {id}\n{text}<|im_end|>
//   tools: <|im_system|>tool_declare<|im_middle|>[{...}]<|im_end|>
//   calls: <|tool_calls_section_begin|><|tool_call_begin|>{id}
//          <|tool_call_argument_begin|>{json}<|tool_call_end|>
//          <|tool_calls_section_end|>
// Call ids on the wire are "functions.{name}:{index}" (the only place the
// function name appears). Kimi-K2.7-Code is thinking-only: every assistant
// turn carries <think>…</think>, and the generation prompt ends with <think>.
// Multimodal content is out of scope (the vision tower is stripped).

#include <cstdint>
#include <string>
#include <string_view>
#include <vector>

#include "lsxcommon/chat_conversation.h"
#include "lsxcommon/tokenizer.h"

namespace lsxcommon {

// Appends @p text to @p out_ids, injecting Kimi control sentinels via
// arrow_tokenizer_lookup and BPE-encoding the rest. Returns false when a
// sentinel is missing or the encoder fails.
bool AppendKimiIds(const ArrowTokenizerDb &tok, std::string_view text,
                   std::vector<int32_t> &out_ids);

} // namespace lsxcommon
```

```cpp
#include "lsxcommon/kimi_moe_tool_format.h"

#include <lsxnetwork/json_escape.h>

#include <cstring>
#include <map>
#include <string>

namespace lsxcommon {
namespace {

using lsxnetwork::json::JsonEscape;

constexpr const char *kImSystem = "<|im_system|>";
constexpr const char *kImUser = "<|im_user|>";
constexpr const char *kImAssistant = "<|im_assistant|>";
constexpr const char *kImMiddle = "<|im_middle|>";
constexpr const char *kImEnd = "<|im_end|>";
constexpr const char *kSectionBegin = "<|tool_calls_section_begin|>";
constexpr const char *kSectionEnd = "<|tool_calls_section_end|>";
constexpr const char *kCallBegin = "<|tool_call_begin|>";
constexpr const char *kArgsBegin = "<|tool_call_argument_begin|>";
constexpr const char *kCallEnd = "<|tool_call_end|>";

std::string WireId(const std::string &name, size_t index)
{
    return "functions." + name + ":" + std::to_string(index);
}

// Splits a wire id back into the function name: "functions.x:3" -> "x",
// "x:3" -> "x", "x" -> "x".
std::string NameFromWireId(std::string_view id)
{
    constexpr std::string_view prefix = "functions.";
    if (id.rfind(prefix, 0) == 0) id.remove_prefix(prefix.size());
    const size_t colon = id.rfind(':');
    return std::string(colon == std::string_view::npos ? id : id.substr(0, colon));
}

void RenderTools(const std::vector<ToolSpec> &tools, std::string &out)
{
    out += kImSystem;
    out += "tool_declare";
    out += kImMiddle;
    out += '[';
    for (size_t i = 0; i < tools.size(); ++i) {
        if (i) out += ',';
        out += R"({"type":"function","function":{"name":")";
        out += JsonEscape(tools[i].name);
        out += R"(","description":")";
        out += JsonEscape(tools[i].description);
        out += R"(","parameters":)";
        out += tools[i].parameters_json.empty() ? "{}" : tools[i].parameters_json;
        out += "}}";
    }
    out += ']';
    out += kImEnd;
}

} // namespace

bool RenderKimiConversation(const ChatConversation &conv, std::string &out)
{
    out.clear();
    if (!conv.tools.empty()) RenderTools(conv.tools, out);

    // Client tool_call_id -> wire id, so a following "tool" message can echo
    // the id the model saw.
    std::map<std::string, std::string> wire_ids;

    for (const ChatMessage &m : conv.messages) {
        if (m.role == "tool") {
            auto it = wire_ids.find(m.tool_call_id);
            const std::string id =
                it == wire_ids.end() ? m.tool_call_id : it->second;
            out += kImSystem;
            out += "tool";
            out += kImMiddle;
            out += "## Return of " + id + "\n";
            out += m.content;
            out += kImEnd;
            continue;
        }
        if (m.role == "system") {
            out += kImSystem + std::string("system") + kImMiddle;
            out += m.content;
            out += kImEnd;
            continue;
        }
        if (m.role == "user") {
            out += kImUser + std::string("user") + kImMiddle;
            out += m.content;
            out += kImEnd;
            continue;
        }
        if (m.role != "assistant") return false;

        out += kImAssistant + std::string("assistant") + kImMiddle;
        out += "<think></think>";
        out += m.content;
        if (!m.tool_calls.empty()) {
            out += kSectionBegin;
            for (size_t i = 0; i < m.tool_calls.size(); ++i) {
                const ToolCall &c = m.tool_calls[i];
                const std::string wire = WireId(c.name, i);
                if (!c.id.empty()) wire_ids[c.id] = wire;
                out += kCallBegin + wire + kArgsBegin;
                out += c.arguments_json.empty() ? "{}" : c.arguments_json;
                out += kCallEnd;
            }
            out += kSectionEnd;
        }
        out += kImEnd;
    }
    out += kImAssistant + std::string("assistant") + kImMiddle + "<think>";
    return true;
}

std::vector<ToolCall> ParseKimiToolCalls(std::string_view text)
{
    std::vector<ToolCall> calls;
    size_t pos = 0;
    while (true) {
        const size_t b = text.find(kCallBegin, pos);
        if (b == std::string_view::npos) break;
        const size_t id_begin = b + std::strlen(kCallBegin);
        const size_t a = text.find(kArgsBegin, id_begin);
        if (a == std::string_view::npos) break;
        const size_t e = text.find(kCallEnd, a + std::strlen(kArgsBegin));
        if (e == std::string_view::npos) break;
        ToolCall c;
        c.id = std::string(text.substr(id_begin, a - id_begin));
        c.name = NameFromWireId(c.id);
        c.arguments_json = std::string(
            text.substr(a + std::strlen(kArgsBegin),
                        e - (a + std::strlen(kArgsBegin))));
        if (!c.name.empty()) calls.push_back(std::move(c));
        pos = e + std::strlen(kCallEnd);
    }
    if (calls.empty()) {
        calls = tools::ParseToolCallsText(text);
        if (calls.empty()) calls = tools::ParseBareToolCallJson(text);
    }
    return calls;
}

bool AppendKimiIds(const ArrowTokenizerDb &tok, std::string_view text,
                   std::vector<int32_t> &out_ids)
{
    static const char *const kSentinels[] = {
        kImSystem, kImUser, kImAssistant, kImMiddle, kImEnd,
        kSectionBegin, kSectionEnd, kCallBegin, kArgsBegin, kCallEnd,
        "<think>", "</think>", "tool_declare"};
    out_ids.clear();
    size_t pos = 0;
    while (pos < text.size()) {
        size_t best = std::string_view::npos;
        const char *marker = nullptr;
        for (const char *s : kSentinels) {
            const size_t at = text.find(s, pos);
            if (at < best) { best = at; marker = s; }
        }
        const size_t end = best == std::string_view::npos ? text.size() : best;
        if (end > pos) {
            std::vector<int32_t> ids(text.size() - pos + 16);
            size_t cap = ids.size();
            const int32_t n = arrow_tokenizer_encode(
                tok, text.data() + pos, end - pos, ids.data(), &cap);
            if (n < 0) return false;
            // arrow_tokenizer_encode prepends bos_id when it is >= 0. The
            // Kimi policy sets bos = -1, but a fixture without the policy
            // would inject it mid-stream — strip it defensively.
            const size_t skip =
                (tok.bos_id >= 0 && n > 0 && ids[0] == tok.bos_id) ? 1 : 0;
            out_ids.insert(out_ids.end(), ids.begin() + skip, ids.begin() + n);
        }
        if (marker == nullptr) break;
        const int32_t id =
            arrow_tokenizer_lookup(tok, marker, std::strlen(marker));
        if (id < 0) return false;
        out_ids.push_back(id);
        pos = best + std::strlen(marker);
    }
    return true;
}

} // namespace lsxcommon
```

```cpp
bool KimiK27CodeModel::BuildConversationInputIds(
    const ArrowTokenizerDb &tok, const ChatConversation &conv,
    std::vector<int32_t> &out_ids) const
{
    std::string text;
    if (!RenderKimiConversation(conv, text)) return false;
    if (arrow_tokenizer_lookup(tok, "<|im_user|>", 12) < 0)
        return BuildPromptInputIds(tok, text, out_ids);
    return AppendKimiIds(tok, text, out_ids);
}

bool KimiK27CodeModel::ParseToolCalls(
    const ArrowTokenizerDb &tok, const std::vector<int32_t> &gen_tokens,
    std::vector<ToolCall> &out_calls) const
{
    out_calls.clear();
    if (gen_tokens.empty()) return true;
    std::string text(4096, '\0');
    size_t cap = text.size();
    for (int attempt = 0; attempt < 8; ++attempt) {
        const int rc = arrow_tokenizer_decode(tok, gen_tokens.data(),
                                              gen_tokens.size(), text.data(),
                                              &cap);
        if (rc == 0) {
            text.resize(cap);
            out_calls = ParseKimiToolCalls(text);
            return true;
        }
        if (rc == -2 && cap > text.size() && cap <= (1u << 20)) {
            text.resize(cap);
            continue;
        }
        break;
    }
    return false;
}
```

- [ ] **Step 2: Write the oracle tool**

`tools/kimi_layer_oracle.py` — loads tensors lazily with `safetensors.safe_open`,
dequantizes only the routed experts the router selects, and runs the vendored

```python
#!/usr/bin/env python3
"""Compare the engine's per-layer hidden states against the checkpoint's own
reference modules, with source weights (BF16 attention, INT4 experts
dequantized on demand). The engine dump comes from KIMI_DUMP_LAYERS.

Usage:
  KIMI_DUMP_LAYERS=0,1,30,60 ./build/.../lsxbenchmark --bench ...   # or the test
  python3 tools/kimi_layer_oracle.py --model /mnt/fast_storage/lsx_model/moonshotai/Kimi-K2.7-Code \
      --ids "The capital of France is" --dump-dir /tmp
"""
import argparse
import glob
import json
import os
import struct

import torch
from safetensors import safe_open
from transformers import AutoTokenizer


class Shards:
    def __init__(self, model_dir):
        self.files = sorted(glob.glob(os.path.join(model_dir, "*.safetensors")))
        self.index = {}
        for path in self.files:
            with safe_open(path, framework="pt", device="cpu") as f:
                for name in f.keys():
                    self.index[name] = path

    def get(self, name, dtype=torch.bfloat16):
        with safe_open(self.index[name], framework="pt", device="cpu") as f:
            return f.get_tensor(name).to(dtype)


def dequant_int4(packed, scale, cols):
    # compressed-tensors pack-quantized symmetric int4: unsigned nibble with
    # implicit zero point 8 (offset binary), low nibble first.
    words = packed.to(torch.int32)
    shifts = torch.arange(8, dtype=torch.int32) * 4
    nib = (words.unsqueeze(-1) >> shifts) & 0xF
    vals = (nib.reshape(words.shape[0], cols).to(torch.float32) - 8.0)
    return vals * scale.repeat_interleave(32, dim=1).to(torch.float32)


def rms_norm(x, w, eps):
    v = x.to(torch.float32)
    v = v * torch.rsqrt(v.pow(2).mean(-1, keepdim=True) + eps)
    return (v * w.to(torch.float32)).to(torch.bfloat16)


def yarn_inv_freq(dim=64, base=50000.0, factor=64.0, orig=4096.0,
                  beta_fast=32.0, beta_slow=1.0):
    import math
    def corr(rot):
        return dim * math.log(orig / (rot * 2 * math.pi)) / (2 * math.log(base))
    lo, hi = math.floor(corr(beta_fast)), math.ceil(corr(beta_slow))
    i = torch.arange(0, dim, 2, dtype=torch.float32)
    extra = 1.0 / (base ** (i / dim))
    inter = extra / factor
    ramp = torch.clamp((i / 2 - lo) / (hi - lo), 0.0, 1.0)
    rf = 1.0 - ramp
    return inter * rf + extra * ramp


def rope(x, positions, inv):
    # Adjacent pairs, matches the reference apply_rotary_pos_emb.
    out = x.clone().to(torch.float32)
    for p in range(inv.numel()):
        th = positions.to(torch.float32) * inv[p]
        c, s = torch.cos(th)[:, None], torch.sin(th)[:, None]
        a, b = out[:, 2 * p], out[:, 2 * p + 1]
        out[:, 2 * p] = a * c - b * s
        out[:, 2 * p + 1] = a * s + b * c
    return out.to(torch.bfloat16)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--model", required=True)
    ap.add_argument("--ids", required=True)
    ap.add_argument("--dump-dir", default="/tmp")
    ap.add_argument("--layers", default="0,1,30,60")
    args = ap.parse_args()
    want = [int(x) for x in args.layers.split(",")]

    cfg = json.load(open(os.path.join(args.model, "config.json")))["text_config"]
    tok = AutoTokenizer.from_pretrained(args.model, trust_remote_code=True)
    ids = tok(args.ids).input_ids
    seq = len(ids)
    print("ids:", ids)

    sh = Shards(args.model)
    H = cfg["hidden_size"]
    def t(name):
        return sh.get("language_model.model." + name)

    x = t("embed_tokens.weight")[torch.tensor(ids)]
    inv = yarn_inv_freq()
    pos = torch.arange(seq)
    scale = (192 ** -0.5) * (0.1 * __import__("math").log(64.0) + 1.0) ** 2

    for layer in range(0, max(want) + 1):
        p = f"layers.{layer}."
        h = rms_norm(x, t(p + "input_layernorm.weight"), cfg["rms_norm_eps"])
        q = t(p + "self_attn.q_b_proj.weight") @ rms_norm(
            t(p + "self_attn.q_a_proj.weight") @ h,
            t(p + "self_attn.q_a_layernorm.weight"), cfg["rms_norm_eps"])
        q = q.view(seq, cfg["num_attention_heads"], 192)
        q_nope, q_pe = q[..., :128], q[..., 128:]
        kv = t(p + "self_attn.kv_a_proj_with_mqa.weight") @ h
        latent, k_pe = kv[:, :512], kv[:, 512:]
        latent = rms_norm(latent, t(p + "self_attn.kv_a_layernorm.weight"),
                          cfg["rms_norm_eps"])
        kvb = t(p + "self_attn.kv_b_proj.weight") @ latent
        kvb = kvb.view(seq, cfg["num_attention_heads"], 256)
        k_nope, v = kvb[..., :128], kvb[..., 128:]
        heads = cfg["num_attention_heads"]
        # Each row of the flattened q_pe carries its token's position.
        q_pos = pos.repeat_interleave(heads)
        q_pe = rope(q_pe.reshape(seq * heads, 64), q_pos, inv).view(seq, heads, 64)
        k_pe = rope(k_pe, pos, inv)
        out = torch.zeros(seq, cfg["num_attention_heads"], 128,
                          dtype=torch.float32)
        for hd in range(cfg["num_attention_heads"]):
            qh = torch.cat([q_nope[:, hd], q_pe[:, hd]], dim=-1).to(torch.float32)
            kh = torch.cat([k_nope[:, hd], k_pe], dim=-1).to(torch.float32)
            s = qh @ kh.T * scale
            mask = torch.triu(torch.ones(seq, seq, dtype=torch.bool), 1)
            s = s.masked_fill(mask, float("-inf"))
            p_attn = torch.softmax(s, dim=-1)
            out[:, hd] = p_attn @ v[:, hd].to(torch.float32)
        attn = t(p + "self_attn.o_proj.weight") @ out.reshape(seq, 8192).to(torch.bfloat16)
        x = x + attn.to(torch.bfloat16)

        h = rms_norm(x, t(p + "post_attention_layernorm.weight"), cfg["rms_norm_eps"])
        if layer < cfg["first_k_dense_replace"]:
            g = t(p + "mlp.gate_proj.weight") @ h
            u = t(p + "mlp.up_proj.weight") @ h
            d = t(p + "mlp.down_proj.weight") @ (
                torch.nn.functional.silu(g.to(torch.float32)) * u.to(torch.float32)).to(torch.bfloat16)
        else:
            sh_exp = (torch.nn.functional.silu(
                (t(p + "mlp.shared_experts.gate_proj.weight") @ h).to(torch.float32)) *
                (t(p + "mlp.shared_experts.up_proj.weight") @ h).to(torch.float32))
            d = t(p + "mlp.shared_experts.down_proj.weight") @ sh_exp.to(torch.bfloat16)
            logits = (h.to(torch.float32) @
                      t(p + "mlp.gate.weight").to(torch.float32).T)
            scores = torch.sigmoid(logits)
            bias = t(p + "mlp.gate.e_score_correction_bias", torch.float32)
            top = torch.topk(scores + bias, cfg["num_experts_per_tok"], dim=-1,
                             sorted=False).indices
            w = torch.gather(scores, 1, top)
            w = w / (w.sum(-1, keepdim=True) + 1e-20) * cfg["routed_scaling_factor"]
            for e_i in range(top.shape[1]):
                for tok_i in range(seq):
                    e = int(top[tok_i, e_i])
                    base = f"{p}mlp.experts.{e}."
                    gg = dequant_int4(sh.get("language_model.model." + base + "gate_proj.weight_packed"),
                                      sh.get("language_model.model." + base + "gate_proj.weight_scale"),
                                      H)
                    uu = dequant_int4(sh.get("language_model.model." + base + "up_proj.weight_packed"),
                                      sh.get("language_model.model." + base + "up_proj.weight_scale"),
                                      H)
                    dd = dequant_int4(sh.get("language_model.model." + base + "down_proj.weight_packed"),
                                      sh.get("language_model.model." + base + "down_proj.weight_scale"),
                                      2048)
                    y = dd @ (torch.nn.functional.silu(gg @ h[tok_i].to(torch.float32)) *
                              (uu @ h[tok_i].to(torch.float32)))
                    d[tok_i] += (float(w[tok_i, e_i]) * y).to(torch.bfloat16)
        x = x + d.to(torch.bfloat16)

        if layer in want:
            path = os.path.join(args.dump_dir, f"kimi_layer_{layer}.bin")
            if not os.path.exists(path):
                print(f"layer {layer}: NO ENGINE DUMP at {path}")
                continue
            with open(path, "rb") as f:
                n_seq, n_hidden = struct.unpack("<ii", f.read(8))
                raw = f.read(n_seq * n_hidden * 2)
            eng = torch.frombuffer(bytearray(raw), dtype=torch.bfloat16).view(n_seq, n_hidden)
            if n_seq != seq:
                print(f"layer {layer}: seq mismatch engine={n_seq} ref={seq}")
                continue
            ref = x.to(torch.float32)
            err = (eng.to(torch.float32) - ref).norm() / ref.norm()
            cos = torch.nn.functional.cosine_similarity(
                eng.to(torch.float32).reshape(-1), ref.reshape(-1), dim=0)
            print(f"layer {layer}: relL2={float(err):.5f} cos={float(cos):.6f}")


if __name__ == "__main__":
    main()
```
