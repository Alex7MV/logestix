> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ingest Qwen3.8-Flash-Next (qwen4_exp, 125B-A6B MoE) into logestix following the Model Ingestion Protocol, producing a working SelfCheck ("Paris") on RTX 5090.

**Architecture:** Extend the Qwen3.5-35B-A3B MoE pipeline (Approach A): new `qwen_moe_exp_*` files beside the etalon reusing the pinned ring, unified `gpu::ExpertDeviceCache` LRU, NVFP4 TC-batch GEMM chain, and `qwen_gated_delta_kernel`. Net-new CUDA work: GatedResidual hyper-connections, QSA indexer, PLE n-gram inject.

**Tech Stack:** C++20, CUDA 13.3 (Blackwell 120/121-real only), Arrow 25 IPC, simdjson, Catch2 v3, CCCL 3.3.

## Global Constraints

- Target arch ONLY `120-real;121-real` (Blackwell). Hopper/Ada/Ampere forbidden.
- C++20, C23. spdlog for all logging (no `std::cout`/`printf`).
- CUDA object lib flags: `--extended-lambda --expt-relaxed-constexpr -fmad=false` (bit-exact math).
- Single `cudaStream_t` per generation; no `cudaDeviceSynchronize` in the loop.
- Shared-memory RMW: writes only from thread 0 behind `__syncthreads()`.
- Norms/biases/router/shared-gate NEVER quantized (NVFP4 invariant).
- All model paths via `lsx_model_metadata(rel)` / `lsx_model_metadata_root()`; zero hardcoded paths.
- File naming: `<family>_<type>_<role>[_<subsystem>].<ext>`; `qwen_moe_exp_*` fits the guard (family `qwen`, type `moe`, subsystem `exp`? NO — subsystem vocab is `attn|csa|head|moe|pipeline`; use roles only: `qwen_moe_exp_convert` fails the guard → see Task 0 fix: extend subsystem list with `exp`).
- abseil containers forbidden in headers reachable from `.cu` TUs.
- Host tests need `LSX_TEST_ALLOW_SYNTHETIC_GEOMETRY=1`; real-model selfcheck tests use `SKIP_RETURN_CODE 4`.
- Windows: no `_putenv` — use `setenv`/`_putenv_s` under `#ifdef _WIN32`.
- Frozen constants (spec): kHidden=2560, kLayers=48, kNumExperts=512, kTopK=10, kMoeIntermediate=640, kSharedIntermediate=640, kVocab=248320, kHeads=24, kKVHeads=2, kHeadDim=256, hc_count=4, hc_lowrank=320, indexer budget 2048/compress 4, PLE layer 2 (HF layer_idx 1), eos primary 248046, pad/bos 248044.

namespace {
constexpr uint64_t kMask64 = (1ull << 63) - 1ull + (1ull << 63);  // (1<<64)-1
constexpr uint64_t kGamma = 0x9E3779B97F4A7C15ull;
constexpr uint64_t kM1 = 0xBF58476D1CE4E5B9ull;
constexpr uint64_t kM2 = 0x94D049BB133111EBull;
constexpr int64_t kPrime1 = 10007;

uint64_t Splitmix64(uint64_t v)
{
    v += kGamma;
    v ^= v >> 30; v *= kM1;
    v ^= v >> 27; v *= kM2;
    v ^= v >> 31;
    return v;
}
} // namespace

void Splitmix64Multipliers(int64_t seed, int ple_layer_index,
                           int64_t unigram_vocab, int ngram_size,
                           int64_t out[3])
{
    const int64_t max_long = (1ll << 63) - 1;
    const int64_t multiplier_max = max_long / (unigram_vocab > 0 ? unigram_vocab : 1);
    const int64_t half_bound = std::max<int64_t>(1, multiplier_max / 2);
    const int64_t base = seed + kPrime1 * ple_layer_index;
    for (int i = 0; i < ngram_size; ++i) {
        const uint64_t v = static_cast<uint64_t>(base) +
                           kGamma * static_cast<uint64_t>(i + 1);
        out[i] = static_cast<int64_t>(2 * (Splitmix64(v) % static_cast<uint64_t>(half_bound)) + 1);
    }
}

#pragma once

// Host-side bit-exact reference sequences for qwen4_exp parity tests.
// Mirrors HF modeling_qwen4_exp.py float32 intermediates. Host-only.

#include <cmath>
#include <cstddef>

namespace lsxcommon::qwen_moe_exp {
namespace hostref {

// HF RMSNorm: x.float() * rsqrt(mean(x^2 per group) + eps) * (1 + weight).
inline void GroupRmsNorm(const float *x, int n, int group, double eps,
                         const float *weight, float *out)
{
    const int groups = n / group;
    for (int g = 0; g < groups; ++g) {
        double acc = 0.0;
        for (int i = 0; i < group; ++i) {
            const double v = x[g * group + i];
            acc += v * v;
        }
        const double inv = 1.0 / std::sqrt(acc / group + eps);
        for (int i = 0; i < group; ++i)
            out[g * group + i] = static_cast<float>(
                x[g * group + i] * inv * (1.0 + weight[g * group + i]));
    }
}

// Qwen4ExpTextGatedResidual.forward float32 replica.
inline void GatedResidualHost(const float *hyper_input, const float *down_w,
                              const float *up_w, const float *inject_w,
                              int H, int hc, int lowrank, double eps,
                              float *mixed_out, float *inject_out)
{
    const int n = H * hc;
    std::vector<float> normed(n), dn(lowrank), up(n);
    GroupRmsNorm(hyper_input, n, H, eps, /*weight=*/nullptr, normed.data());
    // down: [lowrank] = W_dn @ normed; silu(v / hc)
    for (int r = 0; r < lowrank; ++r) {
        double acc = 0.0;
        for (int i = 0; i < n; ++i) acc += down_w[r * n + i] * normed[i];
        const double v = acc / hc;
        dn[r] = static_cast<float>(v / (1.0 + std::exp(-v)));
    }
    // up: [n] = W_up @ dn; sigmoid
    for (int i = 0; i < n; ++i) {
        double acc = 0.0;
        for (int r = 0; r < lowrank; ++r) acc += up_w[i * lowrank + r] * dn[r];
        up[i] = 1.0f / (1.0f + std::exp(-acc));
    }
    // mixed = mean over hc branches of (up ⊙ normed)
    for (int h = 0; h < H; ++h) {
        double acc = 0.0;
        for (int c = 0; c < hc; ++c)
            acc += static_cast<double>(up[c * H + h]) * normed[c * H + h];
        mixed_out[h] = static_cast<float>(acc / hc);
    }
    // injection weights: 2 * sigmoid(W_inj @ normed / hc)
    for (int c = 0; c < hc; ++c) {
        double acc = 0.0;
        for (int i = 0; i < n; ++i) acc += inject_w[c * n + i] * normed[i];
        inject_out[c] = static_cast<float>(2.0 / (1.0 + std::exp(-acc / hc)));
    }
}

// HF Qwen4ExpTextRMSNorm needs (1 + weight) centering; GroupRmsNorm above
// takes the raw weight. This overload applies HC-norm with its stored weight.
inline void GroupRmsNormWeighted(const float *x, int n, int group, double eps,
                                 const float *weight, float *out)
{
    GroupRmsNorm(x, n, group, eps, weight, out);
}

// PLE gate: gate = sign(s) * sqrt(|s| clamped 1e-6) where
// s = sum_c(key_c * query_c) / sqrt(H).
inline float PleGateHost(const float *key, const float *query, int H, int hc)
{
    double acc = 0.0;
    for (int c = 0; c < hc; ++c)
        for (int h = 0; h < H; ++h)
            acc += static_cast<double>(key[c * H + h]) * query[c * H + h];
    const double s = acc / std::sqrt(static_cast<double>(H));
    const double a = std::max(1e-6, std::fabs(s));
    return static_cast<float>(std::sqrt(a) * (s < 0 ? -1.0 : 1.0));
}

} // namespace hostref
} // namespace lsxcommon::qwen_moe_exp
