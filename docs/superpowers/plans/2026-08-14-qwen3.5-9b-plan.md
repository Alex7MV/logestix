```cpp
void rmsnorm_ref_fp64(const float *x, const float *w, float *out, int n, int hidden, double eps) {
    for (int i = 0; i < n; ++i) {
        double sum = 0;
        for (int j = 0; j < hidden; ++j) sum += x[i * hidden + j] * x[i * hidden + j];
        double rms = sqrt(sum / hidden + eps);
        double inv = 1.0 / rms;
        for (int j = 0; j < hidden; ++j)
            out[i * hidden + j] = static_cast<float>(x[i * hidden + j] * inv * (1.0 + w[j]));
    }
}
```

- [x] **Step 3: Implement RMSNorm offset kernel in qwen_inference.cu**

```cuda
__global__ void rmsnorm_offset_qwen_kernel(const __half *x, const __half *w,
    __half *out, int n, int hidden, double eps)
{
    // Each block processes one row
    int row = blockIdx.x;
    if (row >= n) return;
    const __half *row_x = x + row * hidden;
    __half *row_out = out + row * hidden;

    // Compute sum of squares in shared memory
    __shared__ float ssum;
    if (threadIdx.x == 0) ssum = 0.0f;
    __syncthreads();
    float val = __half2float(row_x[threadIdx.x]);
    atomicAdd(&ssum, val * val);
    __syncthreads();

    float rms = rsqrtf(ssum / hidden + static_cast<float>(eps));
    float wf = __half2float(w[threadIdx.x]);
    row_out[threadIdx.x] = __float2half(val * rms * (1.0f + wf));
}
```

Block size = hidden (assume ≤ 1024). For larger hidden, reduce over blocks.

```cpp
void partial_rope_ref_fp64(const float *q, const float *k, const int *pos,
    float *q_out, float *k_out, int seq, int heads, int head_dim,
    int rotary_dim, double theta) {
    for (int t = 0; t < seq; ++t) {
        double angle = pos[t] / theta;
        for (int h = 0; h < heads; ++h) {
            for (int d = 0; d < rotary_dim; d += 2) {
                double cos_v = cos((double)d * angle);
                double sin_v = sin((double)d * angle);
                int idx = (t * heads + h) * head_dim + d;
                double q1 = q[idx], q2 = q[idx + 1];
                q_out[idx] = q1 * cos_v - q2 * sin_v;
                q_out[idx + 1] = q1 * sin_v + q2 * cos_v;
                double k1 = k[idx], k2 = k[idx + 1];
                k_out[idx] = k1 * cos_v - k2 * sin_v;
                k_out[idx + 1] = k1 * sin_v + k2 * cos_v;
            }
            // Pass through non-rotated dims
            int pass_start = (t * heads + h) * head_dim + rotary_dim;
            std::memcpy(q_out + pass_start, q + pass_start, (head_dim - rotary_dim) * sizeof(float));
            std::memcpy(k_out + pass_start, k + pass_start, (head_dim - rotary_dim) * sizeof(float));
        }
    }
}
```

```cpp
void swiglu_ref_fp64(const float *x, const float *gate_w, const float *up_w,
    const float *down_w, float *out, int batch, int hidden, int intermediate) {
    for (int b = 0; b < batch; ++b) {
        // Silu(gate(x)) * up(x)
        std::vector<float> gated(intermediate);
        for (int i = 0; i < intermediate; ++i) {
            double dot = 0;
            for (int j = 0; j < hidden; ++j) dot += x[b * hidden + j] * gate_w[i * hidden + j];
            double sig = 1.0 / (1.0 + exp(-dot));
            double up = 0;
            for (int j = 0; j < hidden; ++j) up += x[b * hidden + j] * up_w[i * hidden + j];
            gated[i] = static_cast<float>(dot * sig * up);
        }
        // down_proj
        for (int j = 0; j < hidden; ++j) {
            double sum = 0;
            for (int i = 0; i < intermediate; ++i) sum += gated[i] * down_w[j * intermediate + i];
            out[b * hidden + j] = static_cast<float>(sum);
        }
    }
}
```
