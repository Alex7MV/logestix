# Qwen3.6-35B-A3B Ingestion Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to
> implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for
> tracking. NOTE (AGENTS.md, Autonomous Mode): Task-subagents are FORBIDDEN in
> this repo — execute every task inline in the main session.

**Goal:** Интеграция Qwen3.6-35B-A3B (registry `qwen3.6-35b-a3b`, формат
`logestix-qwen36moe`) по Model Ingestion Protocol: общее ядро конвертера с 3.5,
свой frozen-паспорт и модельный класс на общем qwen_moe runtime-graph, TDD,
без новых CUDA-ядер.

**Architecture:** Архитектура 3.6 идентична 3.5 (`qwen3_5_moe`), поэтому:
конвертер — общий `ConvertQwenMoeModelCommon(spec)` + две тонкие обёртки;
модель — shared-база `QwenMoeModelBase` (в `qwen_moe_model.h`) с leaf-классами
`Qwen35MoeModel`/`Qwen36MoeModel` (свой kType, свой frozen-гейт, свой
`Infer` с `LsxFastMathScope`). Loader/kernels/generation-driver — family-инфра,
не меняются.

**Tech Stack:** C++20, CUDA 13.x (Blackwell sm_120/121), Arrow 25 (FetchContent),
simdjson, abseil, Catch2 v3, spdlog, CMake presets/Ninja.

**Spec:** `docs/superpowers/specs/2026-09-07-qwen36-35b-a3b-ingestion-design.md`

// Appends an explicit bos_token_id into the synthetic text_config (mirrors
// the real Qwen3.6-35B-A3B config.json layout).
void PatchConfigBos(const fs::path &dir, int bos_id)
{
    const fs::path cfg = dir / "config.json";
    std::ifstream f(cfg);
    std::stringstream ss;
    ss << f.rdbuf();
    std::string j = ss.str();
    const std::string needle = ",\"hidden_act\":\"silu\"";
    const auto pos = j.find(needle);
    REQUIRE(pos != std::string::npos);
    j.replace(pos, needle.size(),
              ",\"hidden_act\":\"silu\",\"bos_token_id\":" +
                  std::to_string(bos_id));
    std::ofstream g(cfg, std::ios::trunc);
    g << j;
}

std::string ReadIndex(const fs::path &out)
{
    std::ifstream f(out / "logestix" / "model.index.json");
    std::stringstream ss;
    ss << f.rdbuf();
    return ss.str();
}

} // namespace

#include "lsxcommon/qwen_moe_36_convert.h"

#include "qwen_moe_36_constants.h"
#include "lsxcommon/qwen_moe_convert.h"

#include <absl/strings/str_format.h>

#include <string>

namespace lsxcommon::qwen_moe_36 {
namespace {

// Cross-checks parsed config against the 3.6 frozen passport. Returns false
// and fills `why` on mismatch (etalon: qwen_moe_convert.cpp
// check_frozen_geometry).
bool check_frozen_geometry_36(const qwen_moe::QwenMoeTopology &top,
                              std::string &why)
{
    using namespace frozen;
    auto bad = [&why](const char *field, int got, int want) {
        why = absl::StrFormat("%s: got %d, want %d", field, got, want);
        return false;
    };
    if (top.hidden_size != kHidden)
        return bad("hidden_size", top.hidden_size, kHidden);
    if (top.num_layers != kLayers)
        return bad("num_hidden_layers", top.num_layers, kLayers);
    if (top.num_attention_heads != kHeads)

// TDD — Qwen3.6-35B-A3B SelfCheck harness (Model Ingestion Protocol step 8).
// Real-model test: SKIPs when no converted Qwen3.6-35B-A3B is present (CTest
// SKIP_RETURN_CODE 4).
#include <catch2/catch_test_macros.hpp>

#include <cmath>
#include <cstdlib>
#include <cstring>
#include <filesystem>
#include <string>
#include <vector>

#include <lsxcommon/model.h>
#include <lsxcommon/model_factory.h>
#include <lsxcommon/path_registry.h>
#include <lsxcommon/tokenizer.h>

#include "qwen_moe_36_model.h"

namespace {

constexpr int kStressCycles = 3;

int SelfCheckCycles()
{
    if (const char *p = std::getenv("LSX_TEST_QWEN36_SELFCHECK_CYCLES")) {
        const int n = std::atoi(p);
        if (n >= 1) return n;
    }
    return kStressCycles;
}

// Converted root: unified vault convention <model>/logestix.
std::filesystem::path Qwen36ConvertedPath()
{
    if (const char *env = std::getenv("LSX_TEST_QWEN36_MODEL_PATH"))
        return std::filesystem::path(env);
    const char *src = std::getenv("LSX_TEST_QWEN36_SOURCE_PATH");
    return src ? std::filesystem::path(src)
               : std::filesystem::path(lsxcommon::lsx_model_models_dir()) /
                     "Qwen3.6-35B-A3B";
}

std::vector<int32_t> EncodeSelfCheckPrompt(lsxcommon::ArrowTokenizerDb &tok)
{
    const char *prompt = "The capital of France is";
    std::vector<int32_t> ids(512, 0);
    size_t cap = ids.size();
    const int32_t n = lsxcommon::arrow_tokenizer_encode(
        tok, prompt, std::strlen(prompt), ids.data(), &cap);
    if (n <= 0) return {};
    ids.resize(static_cast<size_t>(n));
    return ids;
}

} // namespace

TEST_CASE("Qwen3.6-35B-A3B SelfCheck: 'The capital of France is' -> 'Paris', "
          "stress cycles with bit-exact deterministic logits",
          "[qwen36][model][selfcheck][real-model]")
{
    const std::filesystem::path model_dir = Qwen36ConvertedPath();
    if (!std::filesystem::exists(model_dir / "logestix" / "pinned"))
        SKIP("Qwen3.6-35B-A3B converted model directory not present");

    lsxcommon::ModelInitConfig cfg;
    cfg.model_path = model_dir.string();
    auto m = lsxcommon::LsxModelFactory::Create(cfg);
    REQUIRE(m != nullptr);
    // Auto-detection path: index "model" -> "qwen3.6-35b-a3b".
    REQUIRE(m->Type() == "qwen3.6-35b-a3b");
    REQUIRE(m->Load());

    const std::vector<int32_t> ids = EncodeSelfCheckPrompt(m->Tokenizer());
    REQUIRE(ids.size() > 0);

    lsxcommon::ModelRequest req;
    req.input_ids = ids;
    req.max_gen_tokens = 2;

    auto sc0 = m->SelfCheck(req);
    REQUIRE(sc0.ok);
    REQUIRE(sc0.selfcheck_passed);
    REQUIRE(sc0.first_gen_text == "Paris");
    REQUIRE(!sc0.final_logits.empty());

    const std::vector<float> &ref_logits = sc0.final_logits;
    const int32_t ref_first = sc0.first_gen_tok_id;
    const int cycles = SelfCheckCycles();

    for (int i = 1; i < cycles; ++i) {
        auto sc = m->SelfCheck(req);
        REQUIRE(sc.ok);
        REQUIRE(sc.selfcheck_passed);
        CHECK(sc.first_gen_text == "Paris");
        CHECK(sc.first_gen_tok_id == ref_first);
        CHECK(sc.final_logits.size() == ref_logits.size());
        CHECK(std::memcmp(sc.final_logits.data(), ref_logits.data(),
                          ref_logits.size() * sizeof(float)) == 0);
    }

    m->Unload();
}

TEST_CASE("Qwen3.6-35B-A3B generation scratch persists across SelfCheck cycles",
          "[qwen36][model]")
{
    const std::filesystem::path model_dir = Qwen36ConvertedPath();
    if (!std::filesystem::exists(model_dir / "logestix" / "pinned"))
        SKIP("Qwen3.6-35B-A3B converted model directory not present");

    lsxcommon::ModelInitConfig cfg;
    cfg.model_path = model_dir.string();
    auto m = lsxcommon::LsxModelFactory::Create(cfg);
    REQUIRE(m != nullptr);
    REQUIRE(m->Load());

    const std::vector<int32_t> ids = EncodeSelfCheckPrompt(m->Tokenizer());
    REQUIRE(!ids.empty());

    lsxcommon::ModelRequest req;
    req.input_ids = ids;
    req.max_gen_tokens = 2;

    auto sc0 = m->SelfCheck(req);
    REQUIRE(sc0.ok);

    auto *qm = dynamic_cast<lsxcommon::Qwen36MoeModel *>(m.get());
    REQUIRE(qm != nullptr);
    CHECK(qm->GenScratchForTest().dequant_runs == 1);
    CHECK(qm->GenScratchForTest().arena_allocs <= 1);

    auto sc1 = m->SelfCheck(req);
    REQUIRE(sc1.ok);
    CHECK(qm->GenScratchForTest().dequant_runs == 1);
    CHECK(qm->GenScratchForTest().arena_allocs <= 1);

    REQUIRE(sc1.final_logits.size() == sc0.final_logits.size());
    CHECK(std::memcmp(sc1.final_logits.data(), sc0.final_logits.data(),
                      sc0.final_logits.size() * sizeof(float)) == 0);

    m->Unload();
}
